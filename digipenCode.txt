char* lower( char* s )
{
  for( int i = 0; i < std::strlen( s ); ++i )
    s[i] = std::tolower( s[i] );
  return s;
}

char getLowerChar( char c )
{
  auto uchar = static_cast<unsigned char>( c );
  auto lowered = std::tolower( uchar );
  return static_cast<char>( lowered );
}

std::string& toLower( std::string& s )
{
  for( auto itr = std::begin(s); itr != std::end(s); ++itr )
    *itr = getLowerChar( *itr );
  return s;
}

std::string& toLower( std::string& s )
{
  for( auto& c : s ) // range-based for
    c = getLowerChar(c);
  return s;
}
